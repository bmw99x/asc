# asc (Azure App Service Config TUI) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Terminal UI managing Azure App Service application settings (env vars, Key Vault references, deployment-slot-sticky flags) across configured app groupings and live-discovered slots, with kvt-style staged changes.

**Architecture:** Direct architectural port of `keyvault_tui` (kvt): pydantic config → provider protocol → az-CLI client → Textual app with staged mutations, undo stack, and save-confirm diff. Novel parts: AppSetting model with sticky flag, KV-reference parsing, slot discovery/tabs, JSON-batched `appsettings set` saves, sticky-toggle action, KV-reference builder in the edit modal.

**Tech Stack:** Python 3.13+, uv, hatchling, textual>=1.0, pydantic>=2.0, typer>=0.12; dev: pytest, pytest-asyncio, ruff, ty, textual-dev.

**Spec:** `docs/superpowers/specs/2026-08-24-asc-design.md`. kvt source (read-only reference): `keyvault_tui/src/kvt/`.

```python
import re
from dataclasses import dataclass
from enum import Enum, auto

_KV_PREFIX = re.compile(r"@microsoft\.keyvault\((?P<body>.*)\)\s*$", re.IGNORECASE)
_URI = re.compile(
    r"SecretUri=https://(?P<vault>[^.]+)\.vault\.azure\.net/secrets/(?P<secret>[^/;)]+)",
    re.IGNORECASE,
)
_NAMED = re.compile(r"VaultName=(?P<vault>[^;)]+);SecretName=(?P<secret>[^;)]+)", re.IGNORECASE)


@dataclass(frozen=True)
class KeyVaultRef:
    vault: str
    secret: str
    raw: str


def is_kv_ref(value: str) -> bool:
    return value.lstrip().lower().startswith("@microsoft.keyvault(")


def parse_kv_ref(value: str) -> KeyVaultRef | None:
    m = _KV_PREFIX.search(value.strip())
    if not m:
        return None
    body = m.group("body")
    for pat in (_URI, _NAMED):
        hit = pat.search(body)
        if hit:
            return KeyVaultRef(vault=hit.group("vault"), secret=hit.group("secret"), raw=value)
    return None


def compose_kv_ref(vault: str, secret: str) -> str:
    return f"@Microsoft.KeyVault(SecretUri=https://{vault}.vault.azure.net/secrets/{secret})"
```

`AppSetting` mirrors kvt `EnvVar` (with `slot_setting` instead of `is_multiline`; `kv_ref` as `@property` calling `parse_kv_ref(self.value)`). `ActionKind` adds `TOGGLE_STICKY = auto()`; `Action` gains `slot_setting`/`previous_slot_setting` bools documented in the docstring (SET records both previous value and previous flag; TOGGLE_STICKY is self-inverse).

```python
def list_slots(self) -> list[str]:
    out = self._run(["az", "webapp", "deployment", "slot", "list",
                     "--name", self._app, "--resource-group", self._rg,
                     "--subscription", self._sub, "--query", "[].name", "--output", "json"])
    return json.loads(out)

def list_settings(self, slot: str | None = None) -> list[dict]:
    cmd = ["az", "webapp", "config", "appsettings", "list",
           "--name", self._app, "--resource-group", self._rg,
           "--subscription", self._sub, "--output", "json"]
    return json.loads(self._run(cmd + self._slot_args(slot)))

def set_settings(self, settings: list[dict], slot: str | None = None) -> None:
    with tempfile.NamedTemporaryFile(mode="w", suffix=".json", delete=False) as f:
        json.dump(settings, f)
        tmp = Path(f.name)
    tmp.chmod(0o600)
    try:
        self._run(["az", "webapp", "config", "appsettings", "set",
                   "--name", self._app, "--resource-group", self._rg,
                   "--subscription", self._sub, "--settings", f"@{tmp}"]
                  + self._slot_args(slot))
    finally:
        tmp.unlink(missing_ok=True)

def delete_settings(self, names: list[str], slot: str | None = None) -> None:
    self._run(["az", "webapp", "config", "appsettings", "delete",
               "--name", self._app, "--resource-group", self._rg,
               "--subscription", self._sub, "--setting-names", *names]
              + self._slot_args(slot))

def resolve_kv_secret(self, vault: str, secret: str) -> str:
    out = self._run(["az", "keyvault", "secret", "show", "--vault-name", vault,
                     "--name", secret, "--query", "value", "--output", "tsv"])
    return out.removesuffix("\n")

def _slot_args(self, slot: str | None) -> list[str]:
    return ["--slot", slot] if slot else []
```

Note: 0600 chmod happens after the `with` block closes the file (content flushed) and before `az` runs — the file only ever holds values the user already has access to, but keep it tight anyway.

def _net_changes(self) -> tuple[list[AppSetting], list[str]]:
    """Collapse the undo stack into (upserts, deletes) against the loaded baseline."""
    baseline = {s.key: s for s in self._baseline}          # snapshot taken at load
    current = {s.key: s for s in self._all_settings}
    upserts = [s for k, s in current.items()
               if k not in baseline
               or baseline[k].value != s.value
               or baseline[k].slot_setting != s.slot_setting]
    deletes = [k for k in baseline if k not in current]
    return upserts, deletes
```

`self._baseline: list[AppSetting]` is a deep copy captured when settings load and re-captured after each successful save. Renames fall out naturally (old key in deletes, new key in upserts). On `AzureClientError` during apply: notify with stderr, keep stage and undo stack intact (do NOT clear `dirty`). On success: clear stack, re-capture baseline, `dirty = False`.

def discover(run=None) -> dict[str, dict[str, dict[str, str]]]:
    """Return config dict: group (resource group) → app alias → binding fields."""
    run = run or _az_json
    subs = run(["az", "account", "list", "--query", "[].{id:id, tenantId:tenantId}"])
    config: dict[str, dict[str, dict[str, str]]] = {}
    for sub in subs:
        apps = run(["az", "webapp", "list", "--subscription", sub["id"],
                    "--query", "[].{name:name, rg:resourceGroup}"])
        for app in apps:
            config.setdefault(app["rg"], {})[app["name"]] = {
                "app_name": app["name"],
                "resource_group": app["rg"],
                "subscription_id": sub["id"],
                "tenant_id": sub["tenantId"],
            }
    return config
```
